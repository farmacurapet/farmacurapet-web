import firebase_admin from firebase_admin import credentials, firestore from flask import Flask, render_template, request, redirect, url_for, flash app = Flask(__name__) app.secret_key = 'uma_chave_secreta_qualquer' # Necessário para usar flash messages # --- CONFIGURAÇÃO DO FIREBASE --- # Inicializamos fora das funções para que 'db' seja global try: if not firebase_admin._apps: cred = credentials.Certificate("chave.json") firebase_admin.initialize_app(cred) # ESTA LINHA É A CHAVE: Criamos a variável db aqui fora db = firestore.client() print("✅ Conexão com Firebase estabelecida com sucesso!") except Exception as e: db = None print(f"❌ Erro crítico ao conectar no Firebase: {e}") # --- ROTAS --- @app.route('/') def index(): return render_template('index.html') @app.route('/login', methods=['GET', 'POST']) def login(): if request.method == 'POST': email = request.form.get('email') senha = request.form.get('senha') if db is None: return "Erro: O banco de dados não foi inicializado corretamente.", 500 try: # Agora o 'db' está definido e pode ser usado aqui user_ref = db.collection('usuarios').document(email).get() if user_ref.exists: user_data = user_ref.to_dict() if user_data.get('senha') == senha: return f"Login realizado com sucesso! Bem-vindo, {user_data.get('nome')}." else: return "Senha incorreta.", 401 else: return "Usuário não encontrado.", 404 except Exception as e: return f"Erro na consulta ao banco: {e}", 500 return render_template('login.html') @app.route('/cadastro', methods=['GET', 'POST']) def cadastro(): if request.method == 'POST': nome = request.form.get('nome') email = request.form.get('email') senha = request.form.get('senha') if db is None: return "Erro: Banco de dados offline.", 500 try: db.collection('usuarios').document(email).set({ 'nome': nome, 'email': email, 'senha': senha }) return redirect(url_for('login')) except Exception as e: return f"Erro ao cadastrar: {e}", 500 return render_template('cadastro.html') if __name__ == '__main__': app.run(debug=True)
